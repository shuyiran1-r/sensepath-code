/* SensePath on-demand OV2640 / UART / buttons / DRV2605L / MAX98357A.
 * ESP-IDF 5.5.x. Target: ESP32-S3 N16R8 + red 18-pin OV2640.
 * UART at 115200, ASCII commands + framed JPEG and 8-kHz 16-bit mono PCM.
 * Photo data is sent only on CAPTURE, making it practical to return speech.
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>
#include <stdint.h>

#define SP_UART UART_NUM_0
#define SP_MAX_PCM (512 * 1024)
static const char *TAG = "SensePath";
static SemaphoreHandle_t sp_write_lock;
static i2c_master_dev_handle_t sp_drv = NULL;
static i2s_chan_handle_t sp_tx = NULL;

static void sp_lock(void) { xSemaphoreTake(sp_write_lock, portMAX_DELAY); }
static void sp_unlock(void) { fflush(stdout); xSemaphoreGive(sp_write_lock); }

static void sp_event(const char *name) {
    sp_lock();
    printf("\n---SP-EVENT:%s---\n", name);
    sp_unlock();
}

static void buttons_task(void *arg) {
#if SP_ENABLE_BUTTONS
    const gpio_num_t pins[3] = { SP_BTN_RECOGNIZE, SP_BTN_CONFIRM, SP_BTN_REPEAT };
    const char *names[3] = { "RECOGNIZE", "CONFIRM", "REPEAT" };
    int last[3] = {1,1,1};
    uint32_t last_ms[3] = {0,0,0};
    for (int i = 0; i < 3; i++) {
        gpio_config_t cfg = {
            .pin_bit_mask = 1ULL << pins[i], .mode = GPIO_MODE_INPUT,
            .pull_up_en = GPIO_PULLUP_ENABLE,
            .pull_down_en = GPIO_PULLDOWN_DISABLE,
            .intr_type = GPIO_INTR_DISABLE
        };
        ESP_ERROR_CHECK(gpio_config(&cfg));
    }
    while (true) {
        uint32_t now = (uint32_t)(xTaskGetTickCount() * portTICK_PERIOD_MS);
        for (int i = 0; i < 3; i++) {
            int v = gpio_get_level(pins[i]);
            if (!v && last[i] && now - last_ms[i] > 220) {
                last_ms[i] = now;
                sp_event(names[i]);
            }
            last[i] = v;
        }
        vTaskDelay(pdMS_TO_TICKS(25));
    }
#else
    vTaskDelete(NULL);
#endif
}

static esp_err_t sp_write_reg(uint8_t reg, uint8_t value) {
    uint8_t data[] = {reg, value};
    return i2c_master_transmit(sp_drv, data, sizeof(data), 100);
}

static void sp_haptic(int kind) {
#if SP_ENABLE_HAPTIC
    if (!sp_drv) return;
    /* Internal trigger, ERM waveform library; strength codes 1 / 47 / 14. */
    sp_write_reg(0x01, 0x00);
    sp_write_reg(0x04, kind == 2 ? 47 : kind == 3 ? 14 : 1);
    sp_write_reg(0x05, 0x00);
    sp_write_reg(0x0C, 0x01);
#else
    (void)kind;
#endif
}

static void sp_init_haptic(void) {
#if SP_ENABLE_HAPTIC
    i2c_master_bus_config_t bus_cfg = {
        .clk_source = I2C_CLK_SRC_DEFAULT, .i2c_port = I2C_NUM_1,
        .sda_io_num = SP_HAPTIC_SDA, .scl_io_num = SP_HAPTIC_SCL,
        .glitch_ignore_cnt = 7, .flags.enable_internal_pullup = true
    };
    i2c_master_bus_handle_t bus = NULL;
    if (i2c_new_master_bus(&bus_cfg, &bus) != ESP_OK) { ESP_LOGW(TAG, "Haptic bus failed"); return; }
    i2c_device_config_t dev_cfg = {
        .dev_addr_length = I2C_ADDR_BIT_LEN_7, .device_address = 0x5A,
        .scl_speed_hz = 100000
    };
    if (i2c_master_bus_add_device(bus, &dev_cfg, &sp_drv) != ESP_OK) {
        sp_drv = NULL; ESP_LOGW(TAG, "Haptic device add failed"); return;
    }
    if (i2c_master_probe(bus, 0x5A, 100) != ESP_OK) {
        ESP_LOGW(TAG, "DRV2605L did not ACK 0x5A; disable haptic if not wired");
        sp_drv = NULL; return;
    }
    sp_write_reg(0x01, 0x00);
    sp_write_reg(0x03, 0x01);
    ESP_LOGI(TAG, "DRV2605L ready");
#endif
}

static void sp_init_audio(void) {
#if SP_ENABLE_AUDIO
    i2s_chan_config_t ch = I2S_CHANNEL_DEFAULT_CONFIG(I2S_NUM_AUTO, I2S_ROLE_MASTER);
    if (i2s_new_channel(&ch, &sp_tx, NULL) != ESP_OK) { ESP_LOGW(TAG,"I2S channel failed"); sp_tx=NULL; return; }
    i2s_std_config_t cfg = {
        .clk_cfg = I2S_STD_CLK_DEFAULT_CONFIG(8000),
        .slot_cfg = I2S_STD_PHILIPS_SLOT_DEFAULT_CONFIG(I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_STEREO),
        .gpio_cfg = {
            .mclk=I2S_GPIO_UNUSED, .bclk=SP_I2S_BCLK,
            .ws=SP_I2S_LRCLK, .dout=SP_I2S_DATA, .din=I2S_GPIO_UNUSED
        }
    };
    if (i2s_channel_init_std_mode(sp_tx, &cfg) != ESP_OK ||
        i2s_channel_enable(sp_tx) != ESP_OK) {
        ESP_LOGW(TAG,"I2S init failed"); sp_tx=NULL; return;
    }
    ESP_LOGI(TAG, "MAX98357A I2S initialized (8kHz stereo-mirrored PCM)");
#endif
}

static void sp_tone(void) {
#if SP_ENABLE_AUDIO
    if (!sp_tx) return;
    /* Approximate short alert tone with 8 kHz sample rate; no internet/TTS required. */
    int16_t frames[256 * 2];
    for (int n = 0; n < 256; n++) {
        int16_t value = (n % 20 < 10) ? 7000 : -7000;
        frames[n*2] = frames[n*2+1] = value;
    }
    size_t written;
    i2s_channel_write(sp_tx, frames, sizeof(frames), &written, pdMS_TO_TICKS(300));
#endif
}

static void sp_play_pcm(const int16_t *mono, size_t samples) {
#if SP_ENABLE_AUDIO
    if (!sp_tx) return;
    int16_t stereo[256 * 2];
    for (size_t i = 0; i < samples; ) {
        size_t count = samples - i;
        if (count > 256) count = 256;
        for (size_t j = 0; j < count; j++) {
            stereo[2*j] = mono[i+j];
            stereo[2*j+1] = mono[i+j];
        }
        size_t n = 0;
        i2s_channel_write(sp_tx, stereo, count * 4, &n, pdMS_TO_TICKS(500));
        i += count;
    }
#else
    (void)mono; (void)samples;
#endif
}

static void sp_capture(void) {
    camera_fb_t *fb = esp_camera_fb_get();
    if (!fb || fb->format != PIXFORMAT_JPEG) {
        if (fb) esp_camera_fb_return(fb);
        sp_event("CAPTURE_FAILED"); return;
    }
    size_t alloc_len = 0;
    mbedtls_base64_encode(NULL, 0, &alloc_len, fb->buf, fb->len);
    unsigned char *b64 = heap_caps_malloc(alloc_len + 8, MALLOC_CAP_SPIRAM);
    size_t b64_len = 0;
    if (!b64 || mbedtls_base64_encode(b64, alloc_len+8, &b64_len, fb->buf, fb->len) != 0) {
        free(b64); esp_camera_fb_return(fb);
        sp_event("CAPTURE_FAILED"); return;
    }
    sp_lock();
    printf("\n---JPEG-BEGIN %u---\n", (unsigned)fb->len);
    for (size_t i = 0; i < b64_len; i += 512) {
        size_t n = b64_len-i < 512 ? b64_len-i : 512;
        fwrite(b64+i, 1, n, stdout); fputc('\n', stdout);
    }
    printf("---JPEG-END---\n");
    sp_unlock();
    free(b64);
    esp_camera_fb_return(fb);
}

static bool sp_receive_exact(uint8_t *dest, size_t len) {
    size_t got = 0;
    TickType_t last = xTaskGetTickCount();
    while (got < len) {
        int n = uart_read_bytes(SP_UART, dest+got,
                                len-got > 1024 ? 1024 : len-got,
                                pdMS_TO_TICKS(500));
        if (n > 0) {got += n; last=xTaskGetTickCount();}
        else if (xTaskGetTickCount() - last > pdMS_TO_TICKS(5000)) return false;
    }
    return true;
}

void sensepath_start(void) {
    sp_write_lock = xSemaphoreCreateMutex();
    uart_config_t uc = {
        .baud_rate=115200, .data_bits=UART_DATA_8_BITS,
        .parity=UART_PARITY_DISABLE, .stop_bits=UART_STOP_BITS_1,
        .flow_ctrl=UART_HW_FLOWCTRL_DISABLE, .source_clk=UART_SCLK_DEFAULT
    };
    ESP_ERROR_CHECK(uart_param_config(SP_UART,&uc));
    ESP_ERROR_CHECK(uart_driver_install(SP_UART, 8192, 0, 0, NULL, 0));
    sp_init_haptic();
    sp_init_audio();
    xTaskCreate(buttons_task,"sp_buttons",3072,NULL,5,NULL);
    sp_event("READY");
}

void sensepath_loop(void) {
    char cmd[64]; int used=0;
    for (;;) {
        uint8_t c;
        if (uart_read_bytes(SP_UART,&c,1,pdMS_TO_TICKS(50)) != 1) continue;
        if (c == '\r') continue;
        if (c != '\n') {
            if (used < (int)sizeof(cmd)-1) cmd[used++] = (char)c;
            else used=0;
            continue;
        }
        cmd[used]='\0'; used=0;
        if (!strcmp(cmd,"CAPTURE")) {
            sp_capture();
        } else if (!strcmp(cmd,"BEEP")) {
            sp_tone(); sp_event("BEEP_DONE");
        } else if (!strncmp(cmd,"HAPTIC ",7)) {
            sp_haptic(atoi(cmd+7)); sp_event("HAPTIC_DONE");
        } else if (!strncmp(cmd,"AUDIO ",6)) {
            long len = atol(cmd+6);
            if (len <= 0 || len > SP_MAX_PCM || (len & 1)) {
                sp_event("AUDIO_INVALID"); continue;
            }
            uint8_t *buf = heap_caps_malloc((size_t)len, MALLOC_CAP_SPIRAM);
            if (!buf) { sp_event("AUDIO_NOMEM"); continue; }
            sp_event("AUDIO_READY");
            bool ok=sp_receive_exact(buf,(size_t)len);
            if (ok) {sp_event("AUDIO_BUFFERED"); sp_play_pcm((int16_t*)buf,(size_t)len/2);}
            free(buf);
            sp_event(ok?"AUDIO_DONE":"AUDIO_TIMEOUT");
        } else if (!strcmp(cmd,"PING")) {
            sp_event("PONG");
        } else if (*cmd) {
            sp_event("BAD_COMMAND");
        }
    }
}

"""Safely overlay SensePath I/O onto the EXISTING proven camera ESP-IDF project.
Idempotent: backup original camera files before first change; require exact anchors.
Does not modify sdkconfig, board pin mapping, camera power settings or data layout.
"""
from pathlib import Path
import shutil
import sys
import re
import time

DEFAULT = Path(r"D:\SensePath\esp32s3-ov2640-uvc")
repo = Path(sys.argv[1]) if len(sys.argv) > 1 else DEFAULT
repo = repo.expanduser()

cmake = repo / "main" / "CMakeLists.txt"
pins = repo / "main" / "cam_pins.h"

if not main.is_file() or not cmake.is_file() or not pins.is_file():
    sys.exit(f"未找到已有 OV2640 工程（不会修改）：{repo}")
current = main.read_text(encoding="utf-8")
if "sensepath_start();" in current:
    print("检测到 SensePath 固件补丁已安装；不重复修改。")
    sys.exit(0)

for name, val in (("CAM_PIN_SIOD", "15"), ("CAM_PIN_SIOC", "16"),
                  ("STREAM_PCLK_DIV", "4"), ("USE_UVC", "0")):
    match = re.search(r"(?m)^\s*#define\s+" + name + r"\s+(-?\w+)", pin_src)
    if not match or match.group(1) != val:
        sys.exit(f"配置 {name} 与你上次成功版不一致，实际为 " +
                 (match.group(1) if match else '未找到') + "。停止修改，请先人工核对。")

checks = {
    '#include "cam_pins.h"': '#include "cam_pins.h"\n#include "sensepath_io.h"',
    '    cam_console_start();':
        '    /* SensePath owns UART input. Camera console disabled to avoid UART0 conflict. */',
    '    ESP_LOGI(TAG, "streaming JPEG over serial (no WiFi configured)");\n'
    '    while (true) {\n'
    '        dump_jpeg_serial();\n'
    '    }':
        '    ESP_LOGI(TAG, "SensePath button-triggered UART protocol starting");\n'
        '    sensepath_start();\n'
        '    sensepath_loop();',

    if current.count(old) != 1:
        sys.exit("摄像头 main.c 结构与已核对参考版本不同；为安全起见停止修改：" + old[:45])

new = current
for old, replacement in checks.items():
    new = new.replace(old, replacement, 1)

cmtext = cmake.read_text(encoding="utf-8")
if 'SRCS "main.c" "cam_console.c"' not in cmtext:
    sys.exit("CMakeLists.txt 与预期不一致，停止修改。")
cmtext = cmtext.replace('SRCS "main.c" "cam_console.c"',
                        'SRCS "main.c" "cam_console.c" "sensepath_io.c"', 1)
cmtext = cmtext.replace('INCLUDE_DIRS ".")',
                        'INCLUDE_DIRS "."\n'
                        '                        PRIV_REQUIRES esp_driver_gpio esp_driver_uart esp_driver_i2c esp_driver_i2s)', 1)

# Back up first; no silent overwrite of saved working project.
backup = repo / ("SensePath_BACKUP_" + time.strftime("%Y%m%d_%H%M%S"))
backup.mkdir()
for src in (main, cmake, pins):
    shutil.copy2(src, backup / src.name)
for name in ("sensepath_io.c", "sensepath_io.h", "sensepath_pins.h"):
    shutil.copy2(source / name, repo / "main" / name)
main.write_text(new, encoding="utf-8")
cmake.write_text(cmtext, encoding="utf-8")
print("已安装 SensePath 固件补丁。")
print("原固件备份：", backup)
print("请先检查（必要时修改）：", repo / 'main' / 'sensepath_pins.h')

def save_item(item):
    try:
        items = json.loads(SHOPPING_FILE.read_text(encoding="utf-8"))
    except (FileNotFoundError, json.JSONDecodeError):
        items = []

def recognise(jpg: bytes, model: str, demo: bool):
    if demo:
        return {"name": "演示商品（模拟结果）", "brand": "未识别", "size": "无法确认",
                "price": "无法确认", "features": "此为离线演示，不代表AI识别结果",

    key = os.environ.get("GEMINI_API_KEY")

        raise RuntimeError('未设置 GEMINI_API_KEY。先运行 set "GEMINI_API_KEY=你的密钥"，或使用 --demo 测试硬件。')
    from google import genai
    from google.genai import types
    client = genai.Client(api_key=key)
    response = client.models.generate_content(

        contents=[PROMPT, types.Part.from_bytes(data=jpg, mime_type="image/jpeg")],

    if not isinstance(obj, dict):
        raise RuntimeError("Gemini 返回不是商品信息对象")
    return {k: str(obj.get(k, "无法确认")) for k in
            ("name", "brand", "size", "price", "features", "speech")}


def wav_to_8k_pcm(path):
    with wave.open(str(path), "rb") as wav:
        nchan, rate, width = wav.getnchannels(), wav.getframerate(), wav.getsampwidth()
        frames = wav.readframes(wav.getnframes())
    if width == 1:
        a = (np.frombuffer(frames, dtype=np.uint8).astype(np.float32) - 128) * 256
    elif width == 2:
        a = np.frombuffer(frames, dtype="<i2").astype(np.float32)
    elif width == 4:
        a = (np.frombuffer(frames, dtype="<i4").astype(np.float32) / 65536)
    else:
        raise ValueError(f"不支持的语音位深: {width * 8}")
    if nchan > 1:
        a = a.reshape(-1, nchan).mean(axis=1)
    if len(a) < 2:
        return b""
    target = max(2, int(round(len(a) * 8000 / rate)))
    x = np.arange(len(a), dtype=np.float64)
    wanted = np.arange(target, dtype=np.float64) * (rate / 8000.0)
    out = np.interp(wanted, x, a, right=0)
    return np.clip(out, -32000, 32000).astype("<i2").tobytes()


def make_wav(text: str):
    # Use Windows built-in SAPI; no external speech service or API key required.
    stamp = str(time.time_ns())
    text_file = HERE / ("_speech_" + stamp + ".txt")
    wav_file = HERE / ("_speech_" + stamp + ".wav")
    text_file.write_text(text[:250], encoding="utf-8")
    script = HERE / "synthesize.ps1"
    try:
        proc = subprocess.run(
            ["powershell.exe", "-NoProfile", "-ExecutionPolicy", "Bypass",
             "-File", str(script), "-TextPath", str(text_file), "-OutPath", str(wav_file)],
            capture_output=True, text=True, timeout=45,
        )
        if proc.returncode != 0 or not wav_file.exists():
            raise RuntimeError((proc.stderr or proc.stdout or "Windows 语音合成失败")[-350:])
        return wav_file
    finally:
        text_file.unlink(missing_ok=True)


class Board:
    def __init__(self, port, baud, events):
        self.ser = serial.Serial(port, baud, timeout=0.2, write_timeout=30)
        self.ser.dtr = False
        self.ser.rts = False
        self.events = events
        self.send_lock = threading.Lock()
        self.stop = False
        self.audio_ready = threading.Event()
        self.audio_done = threading.Event()
        threading.Thread(target=self.reader, daemon=True).start()

    def cmd(self, command):
        with self.send_lock:
            self.ser.write((command + "\n").encode("ascii"))
            self.ser.flush()

    def reader(self):
        pending = bytearray()
        inside = False
        expected = 0
        chunks = []
        while not self.stop:
            try:
                block = self.ser.read(4096)
            except (serial.SerialException, OSError) as exc:
                self.events.put(("ERROR", str(exc)))
                return
            if not block:
                continue
            pending.extend(block)
            while b"\n" in pending:
                raw, _, rest = pending.partition(b"\n")
                pending = bytearray(rest)
                line = raw.strip()
                if inside:
                    if line == b"---JPEG-END---":
                        try:
                            jpg = base64.b64decode(b"".join(chunks), validate=True)
                            if len(jpg) != expected or not jpg.startswith(b"\xff\xd8") or not jpg.endswith(b"\xff\xd9"):
                                raise ValueError("JPEG 长度或标记无效")
                            old = ImageFile.LOAD_TRUNCATED_IMAGES
                            ImageFile.LOAD_TRUNCATED_IMAGES = False
                            try:
                                with Image.open(BytesIO(jpg)) as im:
                                    im.load()
                                    if im.width < 160:
                                        raise ValueError("图像过小")
                            finally:
                                ImageFile.LOAD_TRUNCATED_IMAGES = old
                            self.events.put(("FRAME", jpg))
                        except Exception as exc:
                            self.events.put(("ERROR", "JPEG 数据异常：" + str(exc)))
                        inside = False
                        chunks = []
                        continue
                    if line.startswith(b"---JPEG-BEGIN"):
                        inside = False; chunks = []
                    else:
                        chunks.append(line)
                        if sum(map(len, chunks)) > 2_000_000:
                            inside = False; chunks = []
                        continue
                bm = BEGIN.match(line)
                if bm:
                    expected = int(bm.group(1))
                    inside = True
                    chunks = []
                    continue
                em = EVT.match(line)
                if em:
                    event = em.group(1).decode()
                    if event == "AUDIO_READY":
                        self.audio_ready.set()
                    elif event in ("AUDIO_DONE", "AUDIO_TIMEOUT", "AUDIO_INVALID"):
                        self.audio_done.set()
                    else:
                        self.events.put(("DEVICE", event))
            if len(pending) > 2_000_000:
                pending.clear()

    def send_pcm(self, pcm: bytes):
        if not pcm:
            return
        self.audio_ready.clear()
        self.audio_done.clear()
        self.cmd(f"AUDIO {len(pcm)}")
        if not self.audio_ready.wait(5):
            raise RuntimeError("ESP32 没有返回 AUDIO_READY（检查 MAX98357A 固件/串口）")
        with self.send_lock:
            # Raw audio bytes, no newline or Base64.
            for i in range(0, len(pcm), 1024):
                self.ser.write(pcm[i:i+1024])
            self.ser.flush()
        if not self.audio_done.wait(65):
            raise RuntimeError("ESP32 没有返回 AUDIO_DONE")

    def close(self):
        self.stop = True
        self.ser.close()


def keyboard(events):
    while True:
        try:
            k = input().strip().lower()
        except EOFError:
            return
        events.put(("KEY", k))
        if k == "q": return


def main():
    p = argparse.ArgumentParser(description="SensePath wearable end-to-end prototype")
    p.add_argument("--port", default="COM6")
    p.add_argument("--baud", type=int, default=115200)
    p.add_argument("--model", default="gemini-2.5-flash")
    p.add_argument("--demo", action="store_true", help="离线演示，用模拟结果测试整套硬件")
    p.add_argument("--audio", choices=("device", "pc", "both", "none"), default="pc",
                   help="默认PC扬声器；设备接线核对后可用 device")
    a = p.parse_args()
    events = queue.Queue()
    try:
        board = Board(a.port, a.baud, events)
    except Exception as exc:
        raise SystemExit(f"COM 串口打开失败：{exc}\n先关闭 live_view.py、idf.py monitor 和 Arduino 串口监视器")
    threading.Thread(target=keyboard, args=(events,), daemon=True).start()
    print("\nSensePath started. Buttons: REC / CONFIRM / REPEAT. Keyboard: R / C / P / Q")
    print("串口 COM 正在等待。设备可拍照识别；不需要打开原 live_view.py。")
    print("模式：", "离线模拟" if a.demo else "Gemini 真实识别", "; 音频：", a.audio)
    last_product = None
    last_speech = ""
    busy = False
    pending = False

    def speech(text):
        def run():
            if a.audio == "none": return
            wav = None
            try:
                wav = make_wav(text)
                if a.audio in ("pc", "both"):
                    import winsound
                    winsound.PlaySound(str(wav), winsound.SND_FILENAME)
                if a.audio in ("device", "both"):
                    board.send_pcm(wav_to_8k_pcm(wav))
            except Exception as exc:
                print("[语音反馈错误]", exc)
            finally:
                if wav: wav.unlink(missing_ok=True)
        threading.Thread(target=run, daemon=True).start()

    def recognize_worker(jpg):
        path = CAPTURES / f"product_{datetime.now().strftime('%Y%m%d_%H%M%S')}.jpg"
        path.write_bytes(jpg)
        print("[拍照]", path)
        try:
            product = recognise(jpg, a.model, a.demo)
            events.put(("RECOGNIZED", product))
        except Exception as exc:
            events.put(("ERROR_RECOGNIZE", str(exc)))

    try:
        while True:
            kind, data = events.get()
            if kind == "KEY" and data == "q": break
            if kind == "ERROR":
                print("[接收异常]", data)
                pending=False; busy=False
            if (kind == "KEY" and data == "r") or (kind == "DEVICE" and data == "RECOGNIZE"):
                if busy or pending:
                    print("[提示] 当前识别尚未结束")
                else:
                    print("[按键] 拍摄商品中……")
                    pending = True
                    board.cmd("HAPTIC 1")
                    board.cmd("CAPTURE")
            elif kind == "FRAME":
                if pending:
                    pending = False
                    busy = True
                    threading.Thread(target=recognize_worker, args=(data,), daemon=True).start()
            elif kind == "RECOGNIZED":
                busy=False
                last_product=data
                last_speech=data["speech"]
                print("\n======= Gemini 识别结果 =======")
                print(json.dumps(data, ensure_ascii=False, indent=2))
                print("===============================")
                board.cmd("HAPTIC 1")
                speech(last_speech)
                print("按 C / 第二按键确认加入购物清单；按 P / 第三按键重播。")
            elif kind == "ERROR_RECOGNIZE":
                busy=False
                print("[识别失败]", data)
                board.cmd("HAPTIC 2")
                speech("识别失败，请重新对准商品拍摄")
            elif (kind == "KEY" and data == "c") or (kind == "DEVICE" and data == "CONFIRM"):
                if last_product and not busy:
                    n=save_item(last_product)
                    print(f"[已确认] 购物清单当前有 {n} 条记录；商品：{last_product['name']}")
                    board.cmd("HAPTIC 3")
                    speech("已确认，加入购物清单")
                else:
                    print("[提示] 请先完成商品识别")
            elif (kind == "KEY" and data == "p") or (kind == "DEVICE" and data == "REPEAT"):
                if last_speech: speech(last_speech)
                else: print("[提示] 尚无识别播报可重复")
            elif kind == "DEVICE" and data == "READY":
                print("[ESP32] SensePath firmware ready")
            elif kind == "DEVICE" and data == "CAPTURE_FAILED":
                pending=False; busy=False
                print("[摄像头] CAPTURE_FAILED，请先检查镜头或传感器")
    except KeyboardInterrupt:
        pass
    finally:
        board.close()
        print("SensePath stopped")


if __name__ == "__main__":
    main()

try {
    $zh = $s.GetInstalledVoices() | Where-Object { $_.VoiceInfo.Culture.Name -like 'zh-*' } | Select-Object -First 1
    if ($zh) { $s.SelectVoice($zh.VoiceInfo.Name) }
    $s.Rate = 0
    $s.SetOutputToWaveFile($OutPath)
    $s.Speak($text)
    $s.SetOutputToNull()
} finally { $s.Dispose() }
